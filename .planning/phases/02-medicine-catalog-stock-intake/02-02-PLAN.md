<objective>
Implement the purchase intake backend — the most complex transaction in Phase 2. A single rusqlite transaction atomically creates the purchase header, line items, batch records, AND stock movements. This is the primary stock-increase pathway (D-22).

**Purpose:** Enable the pharmacist/owner to record medicine purchases. Each purchase creates batch-level stock records with expiry tracking and automatically increases available stock. The atomic transaction ensures data integrity — no orphan purchases or missing stock movements.

**Output:**
- Purchase model types (CreatePurchaseDto, PurchaseReceiptDto, PurchaseDto, etc.)
- Purchase repository (insert purchase + items, query by id/list)
- Purchase service (atomic transaction orchestrator calling batch_repo + stock_ledger_service)
- Purchase commands (record_purchase, list_purchases, get_purchase_detail)
- Registered in main.rs
</objective>

**Key insight:** `rusqlite::Transaction` implements `Deref<Target=Connection>`, so repo functions taking `&Connection` can accept `&Transaction` directly. Within the purchase service, we call `db.transaction()` to get a `Transaction`, then pass `&tx` to repo functions.     **1. `src-tauri/src/services/purchase_service.rs`** — The core transaction orchestrator. This implements Pattern 2 from RESEARCH.md exactly:

    ```rust
    use crate::errors::CommandError;
    use crate::models::{CreatePurchaseDto, CreatePurchaseItemDto, PurchaseReceiptDto, PurchaseDetailDto, PurchaseListDto};
    use crate::repository::{purchase_repo, batch_repo, medicine_repo};
    use crate::services::stock_ledger_service;
    use rusqlite::Connection;

    pub fn record_purchase(
        db: &Connection,
        payload: &CreatePurchaseDto,
        user_id: i64,
    ) -> Result<PurchaseReceiptDto, CommandError> {
        // --- PHASE 1: Validate ALL items BEFORE opening transaction ---
        if payload.items.is_empty() {
            return Err(CommandError::validation("Purchase must have at least one item"));
        }
        if payload.supplier_id <= 0 {
            return Err(CommandError::validation("Invalid supplier"));
        }
        if !["Paid", "Pending", "Partial"].contains(&payload.payment_status.as_str()) {
            return Err(CommandError::validation("Invalid payment status"));
        }

        for (i, item) in payload.items.iter().enumerate() {
            if item.quantity <= 0 {
                return Err(CommandError::validation(
                    &format!("Item {}: quantity must be positive", i + 1)
                ));
            }
            if item.purchase_price < 0.0 {
                return Err(CommandError::validation(
                    &format!("Item {}: purchase price cannot be negative", i + 1)
                ));
            }
            // Validate medicine exists and is active
            let medicine = medicine_repo::find_by_id(db, item.medicine_id)?
                .ok_or_else(|| CommandError::not_found(
                    &format!("Medicine ID {} not found", item.medicine_id)
                ))?;
            if !medicine.is_active {
                return Err(CommandError::validation(
                    &format!("Medicine '{}' is deactivated and cannot be purchased", medicine.name)
                ));
            }
        }

        // --- PHASE 2: Open transaction — atomic: ALL succeed or ALL rollback ---
        let tx = db.transaction().map_err(|e| CommandError::internal(
            &format!("Failed to start transaction: {}", e)
        ))?;

        // --- PHASE 3: Calculate total cost server-side (never trust frontend) ---
        let total_cost: f64 = payload.items.iter()
            .map(|i| i.quantity as f64 * i.purchase_price)
            .sum();

        // --- PHASE 4: Insert purchase header ---
        let purchase_id = purchase_repo::insert_purchase(
            &tx,
            payload.supplier_id,
            payload.invoice_number.as_deref(),
            &payload.purchase_date,
            total_cost,
            &payload.payment_status,
            payload.notes.as_deref(),
            user_id,
        )?;

        // --- PHASE 5: For each item: insert purchase_item + batch + stock_movement ---
        for item in &payload.items {
            let line_cost = item.quantity as f64 * item.purchase_price;

            let item_id = purchase_repo::insert_item(
                &tx,
                purchase_id,
                item.medicine_id,
                item.quantity,
                item.purchase_price,
                &item.expiry_date,
                line_cost,
            )?;

            let batch_id = batch_repo::insert(
                &tx,
                item.medicine_id,
                purchase_id,
                item_id,                           // purchase_item_id
                item.purchase_price,
                item.quantity,                     // quantity
                item.quantity,                     // remaining_qty = quantity (D-20)
                &item.expiry_date,
            )?;

            // Link purchase_item to batch for traceability
            purchase_repo::update_item_batch_id(&tx, item_id, batch_id)?;

            // Record positive stock movement (D-22: StockLedgerService as single authority)
            // Pass &tx here — stock_ledger_repo::insert_movement accepts &Connection,
            // and Transaction implements Deref<Target=Connection>, so this works.
            stock_ledger_service::record_movement(
                &tx,
                "purchase",
                item.medicine_id,
                Some(batch_id),
                item.quantity,         // positive delta (stock increase)
                "purchase",
                Some(purchase_id),
                None,                  // reason (not needed for purchase)
                user_id,
            )?;
        }

        // --- PHASE 6: Update purchase total_cost (recalculated from actual inserts) ---
        purchase_repo::update_purchase_total(&tx, purchase_id, total_cost)?;

        // --- PHASE 7: Commit — ALL or NOTHING ---
        tx.commit().map_err(|e| CommandError::internal(
            &format!("Failed to commit purchase transaction: {}", e)
        ))?;

        Ok(PurchaseReceiptDto {
            purchase_id,
            total_cost,
            item_count: payload.items.len() as i64,
        })
    }

    pub fn list_purchases(
        db: &Connection,
    ) -> Result<Vec<PurchaseListDto>, CommandError> {
        let purchases = purchase_repo::find_all(db)?;
        let mut dtos = Vec::new();
        for (purchase, supplier_name) in purchases {
            let item_count = purchase_repo::get_item_count(db, purchase.id)?;
            dtos.push(PurchaseListDto {
                id: purchase.id,
                supplier_name,
                invoice_number: purchase.invoice_number,
                purchase_date: purchase.purchase_date,
                total_cost: purchase.total_cost.unwrap_or(0.0),
                payment_status: purchase.payment_status,
                item_count,
                created_at: purchase.created_at,
            });
        }
        Ok(dtos)
    }

    pub fn get_purchase_detail(
        db: &Connection,
        purchase_id: i64,
    ) -> Result<PurchaseDetailDto, CommandError> {
        let purchase = purchase_repo::find_by_id(db, purchase_id)?
            .ok_or_else(|| CommandError::not_found("Purchase"))?;

        let items_raw = purchase_repo::find_items_by_purchase_id(db, purchase_id)?;
        let mut items = Vec::new();
        for item in items_raw {
            let medicine = medicine_repo::find_by_id(db, item.medicine_id)?
                .ok_or_else(|| CommandError::not_found("Medicine"))?;
            items.push(PurchaseItemDto {
                id: item.id,
                medicine_id: item.medicine_id,
                medicine_name: medicine.name,
                quantity: item.quantity,
                purchase_price: item.purchase_price,
                line_cost: item.line_cost,
                expiry_date: item.expiry_date,
                batch_id: item.batch_id,
            });
        }

        // Get supplier name
        let supplier_name = {
            // Use a quick query or supplier_repo::find_by_id
            use crate::repository::supplier_repo;
            let supplier = supplier_repo::find_by_id(db, purchase.supplier_id)?
                .ok_or_else(|| CommandError::not_found("Supplier"))?;
            supplier.company_name
        };

        // Get user name
        let created_by = {
            use crate::repository::user_repo;
            // user_repo doesn't have find_by_id returning name — if not available, use "User #{id}"
            format!("User {}", purchase.user_id)
        };

        Ok(PurchaseDetailDto {
            id: purchase.id,
            supplier_id: purchase.supplier_id,
            supplier_name,
            invoice_number: purchase.invoice_number,
            purchase_date: purchase.purchase_date,
            total_cost: purchase.total_cost.unwrap_or(0.0),
            payment_status: purchase.payment_status,
            notes: purchase.notes,
            created_by,
            created_at: purchase.created_at,
            items,
        })
    }
    ```
